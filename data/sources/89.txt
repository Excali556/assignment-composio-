  <script>
    // Sierra HTML Renderer and Analytics
    // This script monitors the Sierra chat for new messages, renders HTML content, and tracks first chat
    ;(function () {
      let isProcessing = false
      let hasTrackedFirstChat = false

      function renderHtmlInMessage(messageElement) {
        // Check if this message has already been processed
        if (messageElement.hasAttribute('data-html-rendered')) {
          return
        }

        // Get all text nodes within the message
        const walker = document.createTreeWalker(messageElement, NodeFilter.SHOW_TEXT, null)
        const textNodes = []
        let node

        while ((node = walker.nextNode())) {
          // Check if the text content looks like HTML (contains HTML tags)
          if (node.textContent && /<[^>]+>/.test(node.textContent)) {
            textNodes.push(node)
          }
        }

        // Process each text node that contains HTML
        textNodes.forEach((textNode) => {
          const htmlContent = textNode.textContent

          // Create a temporary container to parse the HTML
          const tempDiv = document.createElement('div')
          tempDiv.innerHTML = htmlContent

          // Create a wrapper div to hold the rendered HTML
          const wrapper = document.createElement('div')
          wrapper.className = 'sierra-html-content'

          // Move all parsed nodes into the wrapper
          while (tempDiv.firstChild) {
            wrapper.appendChild(tempDiv.firstChild)
          }

          // Replace the text node with the rendered HTML
          if (textNode.parentNode) {
            textNode.parentNode.replaceChild(wrapper, textNode)
          }
        })

        // Mark this message as processed
        messageElement.setAttribute('data-html-rendered', 'true')
      }

      function trackFirstChatIfNeeded(shadowRoot) {
        // Only track once per session
        if (hasTrackedFirstChat) {
          return
        }

        // Look for user message bubbles to detect if user has sent a message
        const userMessageBubbles = shadowRoot.querySelectorAll('[part*="chat-bubble-user"]')

        if (userMessageBubbles.length > 0) {
          // User has sent at least one message, track the event
          hasTrackedFirstChat = true

          if (window.analytics?.track) {
            window.analytics.track('Api Docs AI Assistant First Chat Sent')
            console.log('[Segment] Tracked: Api Docs AI Assistant First Chat Sent')
          }
        }
      }

      function processAllMessages() {
        if (isProcessing) return
        isProcessing = true

        try {
          // Find the Sierra chat container
          const sierraContainer = document.querySelector('[data-sierra-chat-container]')

          if (!sierraContainer) {
            isProcessing = false
            return
          }

          // Access the shadow root
          const shadowRoot = sierraContainer.shadowRoot
          if (!shadowRoot) {
            isProcessing = false
            return
          }

          // Track first chat if needed
          trackFirstChatIfNeeded(shadowRoot)

          // Find all message bubbles (both assistant and user messages)
          const messageBubbles = shadowRoot.querySelectorAll(
            '[part*="chat-bubble-assistant"], [part*="chat-bubble"]'
          )

          messageBubbles.forEach((bubble) => {
            renderHtmlInMessage(bubble)
          })
        } catch (error) {
          console.error('Error processing Sierra messages:', error)
        }

        isProcessing = false
      }

      // Set up a MutationObserver to watch for new messages
      function setupObserver() {
        const sierraContainer = document.querySelector('[data-sierra-chat-container]')

        if (!sierraContainer) {
          // Sierra not loaded yet, try again in 500ms
          setTimeout(setupObserver, 500)
          return
        }

        const shadowRoot = sierraContainer.shadowRoot
        if (!shadowRoot) {
          setTimeout(setupObserver, 500)
          return
        }

        // Process existing messages
        processAllMessages()

        // Watch for new messages
        const observer = new MutationObserver((mutations) => {
          let shouldProcess = false

          mutations.forEach((mutation) => {
            if (mutation.addedNodes.length > 0) {
              shouldProcess = true
            }
          })

          if (shouldProcess) {
            // Use setTimeout to debounce and avoid processing too frequently
            setTimeout(processAllMessages, 100)
          }
        })

        // Start observing the shadow root for changes
        observer.observe(shadowRoot, {
          childList: true,
          subtree: true,
        })

        console.log('Sierra HTML renderer initialized')
      }

      // Start trying to set up the observer once the page loads
      if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', setupObserver)
      } else {
        setupObserver()
      }
    })()
  </script>
